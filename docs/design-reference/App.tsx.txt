type GlobalPreferences = {
  locale: Locale;
  setLocale: (locale: Locale) => void;
  currency: Currency;
  setCurrency: (currency: Currency) => void;
  area: "m²" | "sq ft";
  setArea: (area: "m²" | "sq ft") => void;
};
const PreferencesContext = createContext<GlobalPreferences | null>(null);
function usePreferences() {
  const value = useContext(PreferencesContext);
  if (!value) throw new Error("PreferencesContext is missing");
  return value;
}
function displayPrice(price: string, currency: Currency, locale: Locale) {
  const thb = Number(price.replace(/[^\d]/g, ""));
  if (currency === "THB") return null;
  const amount = Math.round(thb * currencies.find((item) => item.code === currency)!.rate);
  return `≈ ${currencySymbols[currency]}${new Intl.NumberFormat(localeTags[locale]).format(amount)}`;
}

function Button({
  children,
  kind = "primary",
  icon,
  onClick,
  className = "",
}: {
  children: React.ReactNode;
  kind?: "primary" | "secondary" | "ghost" | "danger";
  icon?: IconName;
  onClick?: () => void;
  className?: string;
}) {
  return (
    <button className={`btn btn-${kind} ${className}`} onClick={onClick}>
      {children}{icon && <Icon name={icon} size={18} />}
    </button>
  );
}

function GlobalControls({ footer = false }: { footer?: boolean }) {
  const { locale, setLocale, currency, setCurrency, area, setArea } = usePreferences();
  const [languageOpen, setLanguageOpen] = useState(false);
  const [currencyOpen, setCurrencyOpen] = useState(false);
  return (
    <div className={`global-controls ${footer ? "footer-controls" : ""}`}>
      <div className="global-select">
        <button aria-expanded={languageOpen} onClick={() => { setLanguageOpen(!languageOpen); setCurrencyOpen(false); }}><Icon name="globe" size={17} /><span>{locale}</span><Icon name="chevron" size={13} /></button>
        {languageOpen && <div className="global-menu language-menu"><strong>Language</strong>{languages.map(item => <button className={locale === item.code ? "active" : ""} key={item.code} onClick={() => { setLocale(item.code); setLanguageOpen(false); }}><span><b>{item.label}</b><small>{item.name}</small></span>{locale === item.code && <Icon name="check" size={16} />}</button>)}</div>}
      </div>
      <div className="global-select">
        <button aria-expanded={currencyOpen} onClick={() => { setCurrencyOpen(!currencyOpen); setLanguageOpen(false); }}><span>{currency}</span><Icon name="chevron" size={13} /></button>
        {currencyOpen && <div className="global-menu currency-menu"><strong>Reference currency</strong>{currencies.map(item => <button className={currency === item.code ? "active" : ""} key={item.code} onClick={() => { setCurrency(item.code); setCurrencyOpen(false); }}><span>{item.label}</span>{currency === item.code && <Icon name="check" size={15} />}</button>)}<small><Icon name="info" size={12} /> Converted at today’s rate. export default function App() {
  const initial = (location.hash.replace("#", "") as Screen) || "home";
  const [screen, setScreen] = useState<Screen>(["home", "match", "browse", "detail", "offers", "ai", "saved"].includes(initial) ? initial : "home");
  const [voice, setVoice] = useState(false);
  const [invest, setInvest] = useState(false);
  const [locale, setLocale] = useState<Locale>("EN");
  const [currency, setCurrency] = useState<Currency>("USD");
  const [area, setArea] = useState<"m²" | "sq ft">("m²");
  const go = (next: Screen) => { setScreen(next); location.hash = next; scrollTo({ top: 0, behavior: "smooth" }); };
  return (
    <PreferencesContext.Provider value={{ locale, setLocale, currency, setCurrency, area, setArea }}>
    <div className={`app locale-${locale.toLowerCase()}`} lang={localeTags[locale]}>
      {screen !== "match" && <Header screen={screen} go={go} openVoice={() => setVoice(true)} />}
      {screen === "home" && <Home go={go} openVoice={() => setVoice(true)} />}
      {screen === "match" && <SmartMatch go={go} />}
      {screen === "browse" && <Browse go={go} />}
      {screen === "detail" && <PropertyDetail openInvest={() => setInvest(true)} />}
      {screen === "offers" && <Offers />}
      {screen === "ai" && <AiModelCard />}
      {screen === "saved" && <SavedDashboard go={go} />}
      {screen !== "match" && <Footer go={go} />}
      {screen !== "match" && <BottomTabs screen={screen} go={go} openVoice={() => setVoice(true)} />}
      {voice && <VoiceOverlay close={() => setVoice(false)} go={go} />}
      {invest && <InvestModal close={() => setInvest(false)} />}
    </div>
    </PreferencesContext.Provider>
  );
}
